    conjuntos= set()
    for carater in s:
        conjuntos.add(carater)
    if len(conjuntos)== len(s):
        return True
    return False

def main():

def diferenca_simetrica(m,n):
    c1 = m.union(n)
    c2 = m.intersection(n)

    return c1.difference(c2)

def main():
